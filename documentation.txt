Please note that to avoid race conditions, we have added sleep() in some places, due to that you need to wait a few seconds sometimes
USER2 uses non-blocking k_recvfrom to communicate with multiple clients.
Also even if the client is not present, MAX_CLIENT sockets are bound at the beginning only.

USER2 can communicate with multiple clients but one instance of USER1 can only communicate with one thread of USER2.

Thread S:
   Handles transmission and retransmission of packets.
Thread R:
   Handles recieval of packets,transmission of ACKs and recieval of ACKs as well


ksocket.h:
   ktp_header
      type;        data or ack
      seq_num;     sequence number
      rwnd;        size of reciever window at the time of ACK

   ktp_packet:
      ktp_header header;     header 
      char data[MSG_SIZE];   actual message
      uint16_t len;          message length

   swnd_t:
      int wnd_size;                   size of the sender window, a fixed number
      int acked_wnd_size;             size available in the recv_buf, as informed by the ACK 
      uint8_t seq_nums[BUF_SIZE];     an array of sequence numbers, used to track the packets sent but not ACKed yet
      time_t send_time[BUF_SIZE];     an array of send time to track TIMEOUT for the sent but not ACKed packets
      uint8_t next_seq_num;           next sequence number which is assosciated with the packet which comes next into send_buf
      int start;                      starting position in the buffer
      int cnt;                        number of sent but not ACKED packets



   rwnd_t:
      int wnd_size;                   BUF_SIZE - recv_buf.cnt (Note that it is NOT a fixed number (unlike swnd.wnd_size))
      uint8_t exptd_seq;              the next expected packet seq number to detect whether the recieved packet is a correct/out of order/duplicate
      uint8_t last_ack;               the number of the last ACK sent
      uint8_t rcvd_seq[BUF_SIZE];     used to track dup packets, basically this array contains the seq_nums of packets recieved


   s_buf_t:
      ktp_packet msg[BUF_SIZE];       the messages given by the sender APPLICATION to send via KTP
      int head;                       starting position in the buffer
      int tail;                       ending position in the buffer
      int cnt;                        total number of positions in the send_buf containing packets i.e (sent not ACKED + not yet sent) 



   r_buf_t:
      ktp_packet msg[BUF_SIZE];        the messages recieved via KTP which are to be sent to the reciever APPLICATION
      int head;                        starting position in the buffer
      int tail;                        ending position in the buffer
      int cnt;                         total number of positions in the recv_buf containing packets which are not delivered to the APPLICATION yet


   ktp_sock:
      int is_free;                     whether that socket is free and can be assosciated with a new sender/receiver
      pid_t pid;                       the pid which created the socket
      int udp_sockfd;                  corresponding UDP socket descriptor, for communication via UDP by R and S

      struct sockaddr_in src_addr;     source address
      struct sockaddr_in dest_addr;    destination address

      s_buf_t send_buf;                send buffer
      r_buf_t recv_buf;                recv buffer

      swnd_t swnd;                     sender window
      rwnd_t rwnd;                     receiver window

      int nospace;                     nospace = 1 indicates that the recv_buf is full and the S thread has to wait till it gets an ACK from R saying
                                       that it can restart transmission
      int send_ack;                    k_recvfrom sets nospace = 0 and send_ack = 1 and immediately thread R notices this and sends the ACK saying
                                       that the rwnd > 0 now and data transmission can start
      int needs_udp_init;              whether the KTP is assosciated with a UDP or not
      int needs_bind;                  whether the socket is bound with an address or not



sh_mem:
      ktp_sock sockets[MAX_KTP_SOCK];  an array of KTP data structures
      pthread_mutex_t lock;            a shared lock to lock the shared resources
      int init;                        whether the SM is initialized or not




void init();                                                to initialize the shared memory, data structures

int k_sendto(int sockfd,const void *buf,size_t len,int flags,const struct sockaddr *dest_addr,socklen_t addrlen);

int k_recvfrom(int sockfd,void *buf,size_t len,int flags,struct sockaddr *src_addr,socklen_t *addrlen);

int k_close(int sockfd);

int dropmsg(float p);
